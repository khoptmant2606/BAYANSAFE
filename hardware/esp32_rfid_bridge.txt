/*
  BayanSafe ESP32 + RC522 RFID bridge (WiFiManager edition)
  ---------------------------------------------------------
  Reads a card with an MFRC522 module and posts the UID to the BayanSafe
  Node.js server over Wi-Fi. The server forwards the UID to the browser over
  Server-Sent Events, so the website can detect the card during enrollment.

  Wi-Fi is configured through WiFiManager: on first boot (or whenever the saved
  network is unreachable) the ESP32 creates a captive portal. Connect to it
  with your phone/laptop, open the portal page, and enter your Wi-Fi details.
  They are saved to flash and reused on every future boot. No reflashing needed
  when you change networks.

  Wiring (ESP32 <-> RC522):
    RC522 SDA/SS  -> GPIO 5
    RC522 SCK     -> GPIO 18
    RC522 MOSI    -> GPIO 23
    RC522 MISO    -> GPIO 19
    RC522 RST     -> GPIO 22
    RC522 3.3V    -> 3.3V
    RC522 GND     -> GND

  Required Arduino libraries (install via Library Manager):
    - MFRC522      (by GithubCommunity)
    - WiFiManager  (by tzapu)
    - (WiFi and HTTPClient are built into the ESP32 core)

  Board: "ESP32 Dev Module" (or your board) in Arduino IDE.

  First run:
    1. Flash this sketch.
    2. The ESP32 creates an AP named "BayanSafe-Setup".
    3. Connect to that AP with your phone (password: bayansafe).
    4. A captive portal opens. Pick your Wi-Fi network, enter its password,
       and (optionally) set the Server URL and Bridge token.
    5. Save. The ESP32 reboots and connects to your network.

  To reset Wi-Fi settings later: hold the BOOT button (GPIO 0) for 3 seconds
  during startup, or send 'R' over the serial monitor.
*/

#include <WiFi.h>
#include <HTTPClient.h>
#include <SPI.h>
#include <MFRC522.h>
#include <WiFiManager.h>

// ---------------- CONFIGURATION ----------------
// Defaults used the first time (and as fallback if the portal is skipped).
const char* DEFAULT_SERVER_URL = "http://192.168.100.5:3000";
const char* DEFAULT_BRIDGE_TOKEN = "";

// AP shown during setup
const char* AP_NAME     = "BayanSafe-Setup";
const char* AP_PASSWORD = "bayansafe";

// RC522 pins
#define SS_PIN   5
#define RST_PIN  22

// Reset button (BOOT button on most ESP32 dev boards)
#define RESET_PIN 0
// ----------------------------------------------

MFRC522 mfrc522(SS_PIN, RST_PIN);

// Runtime config (filled in by WiFiManager custom parameters)
char serverUrl[96]    = "";
char bridgeToken[96]  = "";

// Debounce: avoid repeatedly submitting the same card while it stays on reader.
String lastUid       = "";
unsigned long lastSentAt = 0;
const unsigned long RESEND_DELAY_MS = 3000;

void connectWiFi() {
  WiFiManager wm;

  // Custom parameters so the server URL and bridge token can be set from
  // the captive portal instead of being hardcoded.
  WiFiManagerParameter serverParam("server", "Server URL", DEFAULT_SERVER_URL, sizeof(serverUrl) - 1);
  WiFiManagerParameter tokenParam("token", "Bridge token (optional)", DEFAULT_BRIDGE_TOKEN, sizeof(bridgeToken) - 1);
  wm.addParameter(&serverParam);
  wm.addParameter(&tokenParam);

  // Reset trigger: BOOT button held low at startup clears saved settings.
  if (digitalRead(RESET_PIN) == LOW) {
    Serial.println("Reset button held: clearing saved Wi-Fi settings.");
    wm.resetSettings();
    delay(1000);
  }

  // Auto-connect; if it fails, start the captive portal.
  wm.setConfigPortalTimeout(180); // 3 minutes to configure
  bool ok = wm.autoConnect(AP_NAME, AP_PASSWORD);

  if (!ok) {
    Serial.println("Failed to connect / configure. Restarting...");
    delay(3000);
    ESP.restart();
  }

  // Save whatever the user entered (or the defaults).
  strncpy(serverUrl, serverParam.getValue(), sizeof(serverUrl) - 1);
  serverUrl[sizeof(serverUrl) - 1] = '\0';
  strncpy(bridgeToken, tokenParam.getValue(), sizeof(bridgeToken) - 1);
  bridgeToken[sizeof(bridgeToken) - 1] = '\0';

  // Fallback to defaults if left empty.
  if (strlen(serverUrl) == 0)    strcpy(serverUrl, DEFAULT_SERVER_URL);
  if (strlen(bridgeToken) == 0)  strcpy(bridgeToken, DEFAULT_BRIDGE_TOKEN);

  Serial.println("WiFi connected.");
  Serial.print("IP: ");
  Serial.println(WiFi.localIP());
  Serial.print("Server URL: ");
  Serial.println(serverUrl);
}

String readUid() {
  String uid = "";
  for (byte i = 0; i < mfrc522.uid.size; i++) {
    if (mfrc522.uid.uidByte[i] < 0x10) uid += "0";
    uid += String(mfrc522.uid.uidByte[i], HEX);
  }
  uid.toUpperCase();
  return uid;
}

void postUid(const String& uid) {
  if (WiFi.status() != WL_CONNECTED) {
    // WiFiManager will reconnect automatically on next autoConnect attempt;
    // here we just bail and retry later.
    Serial.println("WiFi not connected, skipping post.");
    return;
  }

  HTTPClient http;
  String url = String(serverUrl) + "/api/rfid/scan";
  http.begin(url);
  http.addHeader("Content-Type", "application/json");
  if (strlen(bridgeToken) > 0) {
    http.addHeader("x-bridge-token", bridgeToken);
  }

  String payload = "{\"uid\":\"" + uid + "\"}";
  int code = http.POST(payload);

  if (code > 0) {
    Serial.printf("POST %s -> %d\n", url.c_str(), code);
    String response = http.getString();
    Serial.println(response);
  } else {
    Serial.printf("POST failed: %s\n", http.errorToString(code).c_str());
  }
  http.end();
}

void setup() {
  Serial.begin(115200);
  delay(1000);
  Serial.println();
  Serial.println("BayanSafe ESP32 RFID bridge (WiFiManager) starting...");

  pinMode(RESET_PIN, INPUT_PULLUP);

  SPI.begin();
  mfrc522.PCD_Init();
  Serial.println("RC522 initialized.");

  connectWiFi();
}

void loop() {
  // Serial command to reset Wi-Fi settings: send 'R'.
  if (Serial.available() > 0) {
    char c = Serial.read();
    if (c == 'R' || c == 'r') {
      Serial.println("Resetting Wi-Fi settings...");
      WiFiManager wm;
      wm.resetSettings();
      delay(1000);
      ESP.restart();
    }
  }

  // Reconnect if dropped.
  if (WiFi.status() != WL_CONNECTED) {
    delay(1000);
    WiFi.reconnect();
    delay(2000);
    return;
  }

  // Look for a new card
  if (!mfrc522.PICC_IsNewCardPresent()) {
    delay(50);
    return;
  }
  if (!mfrc522.PICC_ReadCardSerial()) {
    delay(50);
    return;
  }

  String uid = readUid();
  Serial.print("Card detected: ");
  Serial.println(uid);

  // Debounce repeated submissions while the card stays on the reader.
  unsigned long now = millis();
  bool sameCard = (uid == lastUid);
  bool tooSoon  = (now - lastSentAt) < RESEND_DELAY_MS;

  if (!(sameCard && tooSoon)) {
    postUid(uid);
    lastUid     = uid;
    lastSentAt  = now;
  } else {
    Serial.println("Skipping duplicate scan (card still on reader).");
  }

  // Halt PICC and stop crypto to be ready for next card
  mfrc522.PICC_HaltA();
  mfrc522.PCD_StopCrypto1();

  delay(200);
}
